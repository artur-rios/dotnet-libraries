{{- /* Inline "Name x.y.z, Name x.y.z" list of the packages in one group. Packages that are not published are named as deferred rather than given a version. Usage: {{< package-versions "data" >}} */ -}} {{- $group := .Get 0 -}} {{- $packages := where hugo.Data.family.packages "group" $group -}} {{- if not $packages -}} {{- errorf "package-versions: no packages in group %q" $group -}} {{- end -}} {{- range $i, $p := $packages -}} {{- if $i }}, {{ end -}} {{ $p.id }} {{ if eq $p.published false }}(deferred){{ else }}{{ $p.version }}{{ end }} {{- end -}}
